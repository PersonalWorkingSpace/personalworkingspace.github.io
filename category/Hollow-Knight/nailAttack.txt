void SpawnFragments(int count, Vector2 offset)
{
    for (int i = 0; i < count; i++)
    {
        Rigidbody2D sample = fragmentSamples[Random.Range(0, fragmentSamples.Length)];

        Vector2 spawnPos = (Vector2)sample.transform.position + Random.insideUnitCircle * spawnJitter;
        Quaternion spawnRot = Quaternion.Euler(0f, 0f, Random.Range(0f, 360f));

        Rigidbody2D rb = Instantiate(sample, spawnPos, spawnRot, null);
        rb.bodyType = RigidbodyType2D.Dynamic;
        rb.gameObject.SetActive(true);

        Launch(rb, spawnPos, offset);
        Destroy(rb.gameObject, fragmentLifetime);
    }
}

}
